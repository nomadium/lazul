# meson cross file for picolibc on the amd64 kernel, used by the Dockerfile
# (Debian packages no picolibc for x86_64 bare metal). It is built with the
# host compiler and the kernel's own code generation flags:
#
#   -fno-pic -fno-PIE -static -nostdlib
#                       linked into the kernel at a fixed address, with no
#                       C library of the host
#   -mno-red-zone       interrupts arrive on the current stack, so code must
#                       not keep data below the stack pointer
#   -fno-stack-protector -U_FORTIFY_SOURCE -fcf-protection=none
#                       distribution defaults that expect a Linux libc or
#                       CPU features the kernel does not set up
[binaries]
c = ['x86_64-linux-gnu-gcc', '-fno-pic', '-fno-PIE', '-static', '-nostdlib', '-mno-red-zone', '-fno-stack-protector', '-U_FORTIFY_SOURCE', '-fcf-protection=none']
ar = 'x86_64-linux-gnu-ar'
as = 'x86_64-linux-gnu-as'
nm = 'x86_64-linux-gnu-nm'
strip = 'x86_64-linux-gnu-strip'
objcopy = 'x86_64-linux-gnu-objcopy'

[host_machine]
system = 'none'
cpu_family = 'x86_64'
cpu = 'x86_64'
endian = 'little'

[properties]
skip_sanity_check = true
link_spec = '--build-id=none'
# Only used by picolibc's own linker script, which the kernel replaces.
default_flash_addr = '0x00100000'
default_flash_size = '0x00400000'
default_ram_addr = '0x00500000'
default_ram_size = '0x00200000'
